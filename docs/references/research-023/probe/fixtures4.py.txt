results = []


def check(name, cond, detail=""):
    results.append((name, bool(cond)))
    print(("PASS " if cond else "FAIL ") + name, detail)


def bundle(pools, c):
    return SnapshotBundle("fx", "synthetic", 1, BlockRef(1, 1, "h", 1), {p.pool_id: p for p in pools}, (c,), "h", "fx")


def step(pid, tin, tout, fid, amount, out):
    return SwapStep(pid, tin, tout, (FundInput(fid, amount),), out)


def trace(ev):
    return [t.to_dict() for t in ev.trace]


def run(b, c, plan, mode="polish", solver="golden", rounds=2, K=2, cap=None):
    ev = evaluate(b, c, plan, O)
    assert ev.status == "ok", ev.error
    with metered_quotes(None) as m:
        r, h, e1 = P.run_case(b, c, trace(ev), mode, solver, 1, rounds, K, cap)
        seam_solver = m.counted
    rep = evaluate(b, c, h.plan, O)
    return ev, r, h, rep, seam_solver

# F16 R3-m1: a time stop anywhere (incl. activation validation) is reported as time, never max_quotes
class TimeAt(P.Ledger):
    def __init__(self, at):
        super().__init__(None)
        self.at = at

    def quote(self, st_, tin, amt):
        if self.n >= self.at:
            raise P.Budget("time")
        return super().quote(st_, tin, amt)
