Writes summary.json (every gate/verdict number with the SHA-256 of the artifact it came
from) and heldout-per-case.csv (one row per held-out case). Run from the repository root."""
import csv, hashlib, json, re, sys
from pathlib import Path

            "/Users/whisker/Work/src/tools/work/mantle-router/router-algorithms-optimizer-artifacts/metis-challenge")

S4 = ROOT / "campaign-s4-446674e-20260928T163229Z"

CX = ROOT / "campaign-ctxsor-446674e-20260929T024043Z"


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def js(p):
    return json.loads(Path(p).read_text())


def records(run_dir):
    return [json.loads(line) for line in (Path(run_dir) / "cases.jsonl").read_text().splitlines() if line]


def loads(path, start, end):
    vals = []
    for line in Path(path).read_text().splitlines():
        t, rest = line.split(" ", 1)
        if start <= t <= end:
            vals.append(float(re.findall(r"[\d.]+", rest)[0]))
    return {"samples_1min": len(vals), "max_1min": max(vals) if vals else None, "samples_above_5.0": sum(v > 5.0 for v in vals)}


def median(xs):
    xs = sorted(xs)
    n = len(xs)
    return None if not n else xs[n // 2] if n % 2 else (xs[n // 2 - 1] + xs[n // 2]) / 2


s2 = js(S2S3 / "s2-diagnostic.json")
s2_classes = {}
for c in s2["cases"]:
    for k, v in ((c.get("diagnostic") or {}).get("classes") or {}).items():
        s2_classes[k] = s2_classes.get(k, 0) + v
s3 = js(S2S3 / "s3-work-gate.json")
s4 = js(S4 / "s4-disposition.json")
h4 = js(H4 / "h4-diagnostic.json")
ho = js(HO / "heldout-analysis.json")

cx = js(CX / "context-analysis.json")

m4_dir = HO / "runs-m4" / ho["runs"]["M4"]["run_id"]
a0 = {r["case_id"]: r for r in records(a0_dir)}
m4 = {r["case_id"]: r for r in records(m4_dir)}
hops = {}
for r in m4.values():
    for k, v in ((r.get("search") or {}).get("chunk_path_hops") or {}).items():
        hops[k] = hops.get(k, 0) + v
a0_hops = {}
for r in a0.values():
    for k, v in ((r.get("search") or {}).get("chunk_path_hops") or {}).items():
        a0_hops[k] = a0_hops.get(k, 0) + v


def work(recs, key):
    xs = [(r.get("search") or {}).get(key) for r in recs]
    xs = [x for x in xs if isinstance(x, int)]
    return {"n": len(xs), "median": median(xs), "min": min(xs) if xs else None, "max": max(xs) if xs else None}

                **{k: ho[k] for k in ("paired_ok_cases", "excluded_from_pairing", "wins", "losses", "ties", "sign_test", "verdict",
