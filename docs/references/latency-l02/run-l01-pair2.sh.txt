#!/bin/bash
# WHI-1504 second (authorized, single) back-to-back L01 pair.
# baseline 299b88a (measurement-only detached checkout) -> its L01-SB,
# then candidate feature-branch HEAD (must equal $CAND) -> its L01-SB.
# Aborts on the first failing step with its real exit status. On INT/TERM/EXIT it first
# terminates and waits for the active measurement child (and its process group), and only
# then restores the feature branch checkout. Never checks out while workers are alive.
set -u -o pipefail
WT=/Users/whisker/Work/src/tools/work/mantle-router/router-algorithms-optimizer-wt/whi-1504
BUNDLE=/Users/whisker/Work/src/tools/work/mantle-router/router-algorithms-optimizer/data/corpus/mantle-5src-101082044/bundle
OUT=data/latency-l02/l01-pair2
MANIFEST=$OUT/pair-manifest.jsonl
BRANCH=feat/whi-1504-empty-cl-spans
BASE=299b88a787b63900229280f79a3d741e5710f169
CAND=4e48b72f6793a4112c64e91cc92da4ca0fbfa8c7
CHILD=""
cd "$WT" || exit 2
mkdir -p "$OUT"
log() { echo "[$(date -u +%FT%TZ)] $* (load $(sysctl -n vm.loadavg))"; }
record() {  # stage status exit_code
  printf '{"t":"%s","stage":"%s","status":"%s","exit":%s,"head":"%s","load":"%s"}\n' \
    "$(date -u +%FT%TZ)" "$1" "$2" "$3" "$(git rev-parse HEAD)" "$(sysctl -n vm.loadavg)" >> "$MANIFEST"
}
stop_child() {
  if [ -n "$CHILD" ] && kill -0 "$CHILD" 2>/dev/null; then
    log "stopping active child $CHILD (SIGTERM, driver finalizes as interrupted)"
    kill -TERM "$CHILD" 2>/dev/null
    for _ in $(seq 1 120); do kill -0 "$CHILD" 2>/dev/null || break; sleep 1; done
    kill -0 "$CHILD" 2>/dev/null && { log "child $CHILD still alive; SIGKILL"; kill -KILL "$CHILD"; }
    wait "$CHILD" 2>/dev/null
  fi
  CHILD=""
  # no measurement worker of this worktree may survive the checkout restore
  for _ in $(seq 1 30); do
    pgrep -f "$WT/.venv/bin/python" >/dev/null || break; sleep 1
  done
}
restore() {
  stop_child
  if pgrep -f "$WT/.venv/bin/python" >/dev/null; then
    log "workers still alive; NOT changing checkout"; record restore refused 1; return
  fi
  git checkout -q "$BRANCH" && log "restored $BRANCH at $(git rev-parse HEAD)"
}
on_signal() { log "signal received"; record signal interrupted 130; exit 130; }
trap on_signal INT TERM
trap restore EXIT
step() {  # name, timeout, command...
  local name=$1 limit=$2; shift 2
  log "$name: start at $(git rev-parse HEAD)"; record "$name" start null
  timeout "$limit" "$@" &
  CHILD=$!
  wait "$CHILD"; local rc=$?
  CHILD=""
  log "$name: exit $rc"
  if [ "$rc" -ne 0 ]; then record "$name" failed "$rc"; exit "$rc"; fi
  record "$name" complete 0
}
[ -z "$(git status --porcelain)" ] || { log "dirty tree; refusing"; record preflight dirty 3; exit 3; }
[ "$(git rev-parse HEAD)" = "$CAND" ] || { log "HEAD is not $CAND"; record preflight wrong_head 4; exit 4; }
git checkout -q --detach "$BASE" || { record checkout_baseline failed 5; exit 5; }
step baseline_l01 7200 uv run python -m benchmark.latency run --protocol config/latency/l01.yaml --bundle "$BUNDLE" --out "$OUT"
step baseline_sb 1800 uv run python -m benchmark.latency sufficient --sufficient config/latency/l01-sufficient-budget.yaml --bundle "$BUNDLE" --out "$OUT"
git checkout -q "$BRANCH" || { record checkout_candidate failed 6; exit 6; }
[ "$(git rev-parse HEAD)" = "$CAND" ] || { record checkout_candidate wrong_head 7; exit 7; }
step candidate_l01 7200 uv run python -m benchmark.latency run --protocol config/latency/l01.yaml --bundle "$BUNDLE" --out "$OUT"
step candidate_sb 1800 uv run python -m benchmark.latency sufficient --sufficient config/latency/l01-sufficient-budget.yaml --bundle "$BUNDLE" --out "$OUT"
record pair complete 0
log done
