import json
import os
import platform
import signal
import subprocess
import time
from pathlib import Path

ROOT = Path('/Users/whisker/Work/src/tools/work/mantle-router/router-algorithms-optimizer')
OUT = Path('/tmp/router-latency-research')
OUT.mkdir(exist_ok=True)
summary = {
    'kind': 'unprofiled independent CLI invocations; exploratory n=3, not p95/SLA evidence',
    'commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
    'platform': platform.platform(), 'cpu_count': os.cpu_count(),
    'loadavg_start': os.getloadavg(), 'runs': [],
}
start_all = time.monotonic()
for index in range(3):
    output_dir = OUT / f'baseline-{index + 1}'
    command = ['uv', 'run', 'python', 'main.py', 'quote', '--bundle',
               'data/corpus/mantle-5src-101082044/bundle', '--profile',
               'config/daily_gross.yaml', '--token-in', 'USDC', '--token-out',
               'USDT', '--amount', '1000', '--details', '--quotes-dir', str(output_dir)]
    remaining = 180 - (time.monotonic() - start_all)
    if remaining <= 0:
        summary['stopped'] = 'research wall-time bound exhausted'
        break
    started = time.perf_counter()
    process = subprocess.Popen(command, cwd=ROOT, stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE, text=True, start_new_session=True)
    try:
        stdout, stderr = process.communicate(timeout=min(90, remaining))
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGTERM)
        try:
            stdout, stderr = process.communicate(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(process.pid, signal.SIGKILL)
            stdout, stderr = process.communicate()
        summary['runs'].append({'index': index + 1, 'error': 'research wrapper timeout; exclude from completed timings'})
        break
    (OUT / f'baseline-{index + 1}.stdout').write_text(stdout)
    (OUT / f'baseline-{index + 1}.stderr').write_text(stderr)
    row = {'index': index + 1, 'command': command, 'exit': process.returncode,
           'six_algorithm_cli_wall_seconds': time.perf_counter() - started,
           'loadavg_end': os.getloadavg()}
    if process.returncode == 0:
        records_path = next(output_dir.glob('*/runs/*/cases.jsonl'))
        records = [json.loads(line) for line in records_path.read_text().splitlines()]
        manifest_path = records_path.parent / 'manifest.json'
        manifest = json.loads(manifest_path.read_text())
        row.update(records_path=str(records_path), manifest_path=str(manifest_path),
                   records=records, manifest=manifest)
    else:
        row['error'] = stderr[-2000:]
    summary['runs'].append(row)
    (OUT / 'baseline-full.json').write_text(json.dumps(summary, indent=2) + '\n')
    if process.returncode:
        break
summary['loadavg_end'] = os.getloadavg()
(OUT / 'baseline-full.json').write_text(json.dumps(summary, indent=2) + '\n')
(OUT / 'baseline.done').write_text('complete\n')
